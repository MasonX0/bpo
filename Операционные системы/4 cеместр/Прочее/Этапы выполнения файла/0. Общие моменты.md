// example.c
#include <stdio.h>
#define SQUARE(x) ((x)*(x))

int main() {
    printf("Square of 5: %d\n", SQUARE(5));
    return 0;
}

```

Типы символов в nm:

```

nm file.o

0000000000000000 B buffer    # B - неинициализированные данные (.bss)
0000000000000000 D counter   # D - инициализированные данные (.data)
0000000000000000 T main      # T - код/функция (.text)
0000000000000000 R MAX_SIZE  # R - read-only данные
                 U printf    # U - неопределенный символ
0000000000000000 W weak_func # W - weak символ
